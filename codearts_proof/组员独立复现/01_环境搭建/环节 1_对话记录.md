```bat
@echo off
rem ============================================================
rem  依赖安装脚本  install.bat
rem  要求：Windows 7+ / Python 3.10 及以上
rem  注意：本文件必须保存为 UTF-8（无 BOM），否则中文提示会乱码
rem ============================================================
chcp 65001 >nul
cd /d "%~dp0"
setlocal

echo ============================================================
echo  步骤 0：环境自检
echo ============================================================

rem ---------- 0.1 依赖清单是否存在 ----------
if not exist "requirements.txt" (
    echo [错误] 当前目录找不到 requirements.txt
    echo        当前目录：%CD%
    pause
    exit /b 1
)

rem ---------- 0.2 定位解释器（必须先过版本门槛） ----------
rem PY     = 最终选定的、版本满足要求的解释器
rem PY_ANY = 找到但不一定满足版本的解释器，仅用于给出准确报错
set "PY="
set "PY_ANY="

rem 优先 py 启动器；-3 只会选 Python 3，再单独校验小版本
py -3 -c "import sys" >nul 2>&1
if not errorlevel 1 (
    set "PY_ANY=py -3"
    py -3 -c "import sys; sys.exit(0 if sys.version_info[:2] >= (3, 10) else 1)" >nul 2>&1
    if not errorlevel 1 set "PY=py -3"
)

rem py 不可用时回退 PATH 里的 python
rem 先用 where 确认存在，避免命中 WindowsApps 应用别名而弹出 Microsoft Store
if not defined PY (
    where python >nul 2>&1
    if not errorlevel 1 (
        set "PY_ANY=python"
        python -c "import sys; sys.exit(0 if sys.version_info[:2] >= (3, 10) else 1)" >nul 2>&1
        if not errorlevel 1 set "PY=python"
    )
)

if defined PY goto :python_ready
if defined PY_ANY goto :version_too_low
goto :python_missing

:version_too_low
rem 把真实版本号取出来提示；for 放在分支里，避免嵌在 if 括号中解析出错
for /f "delims=" %%v in ('%PY_ANY% -c "import sys;print(sys.version.split()[0])"') do set "PYVER=%%v"
echo [错误] 当前 Python 版本为 %PYVER%，低于要求的 3.10。
echo        请升级 Python 后重新运行本脚本。
pause
exit /b 1

:python_missing
echo [错误] 未找到任何可用的 Python 解释器。
echo        请安装 Python 3.10+，安装时务必勾选 "Add Python to PATH"。
echo        下载地址：https://www.python.org/downloads/windows/
pause
exit /b 1

:python_ready
echo [info] 使用解释器：%PY%
rem 打印真实路径，便于和 run.bat 对照，排查"装到别的解释器"问题
%PY% -c "import sys; print('       ' + sys.executable)"

rem ---------- 0.3 pip 是否可用 ----------
%PY% -m pip --version >nul 2>&1
if errorlevel 1 (
    echo [错误] 该解释器没有可用的 pip，请先修复：
    echo        %PY% -m ensurepip --upgrade
    pause
    exit /b 1
)

rem ---------- 1. 安装依赖 ----------
echo.
echo [1/2] 正在安装依赖：fastapi / uvicorn / openai / python-dotenv ...
rem 保留 pip 原始输出（不要 >nul），报错信息是排查唯一线索
rem --no-input 防止 pip 停在交互式输入上导致窗口看似卡死
%PY% -m pip install --disable-pip-version-check --no-input -r requirements.txt
if errorlevel 1 goto :pip_failed

rem ---------- 2. 导入自检 ----------
rem pip 返回 0 不等于可用：可能装进别的解释器，或只装了一半
echo [2/2] 正在校验依赖能否导入 ...
%PY% -c "import fastapi, uvicorn, openai, dotenv" >nul 2>&1
if errorlevel 1 goto :import_failed

echo.
echo 安装完成！
echo   启动：双击 run.bat，浏览器打开 http://127.0.0.1:8000
echo   如需接入真实模型：把 .env.example 复制为 .env 并填写 API Key
pause
exit /b 0

:pip_failed
echo.
echo [错误] 依赖安装失败（pip 返回非 0 退出码）。
echo        排查顺序：
echo        1. 网络问题，换国内镜像重试：
echo           %PY% -m pip install -r requirements.txt -i https://pypi.tuna.tsinghua.edu.cn/simple
echo        2. pip 损坏，修复后重试：
echo           %PY% -m ensurepip --upgrade
echo        3. 权限不足（装在 Program Files 下时需要管理员权限）：
echo           以管理员身份运行本脚本，或改用 --user 参数安装
pause
exit /b 1

:import_failed
echo.
echo [错误] 依赖安装看似成功，但导入自检失败。
echo        常见原因：install.bat 与 run.bat 用的不是同一套 Python。
echo        可执行 %PY% -c "import sys;print(sys.executable)" 核对路径。
pause
exit /b 1
```

## 一、现有脚本的缺陷

先说明一点：你贴的这两个脚本**其实已经实现了**「版本检查 / pip 失败报错 / 端口占用检测」三项，但每项都还有漏洞：

**install.bat**
1. `install.bat:13-19` 先按 `py -3` → `python` 的顺序直接选定解释器，直到第 30 行才校验版本。如果机器上 `py -3` 指向 3.9、而 `python` 是 3.12，`install.bat:31` 会直接报版本错误退出，**不会回退去试另一个解释器**。
2. `install.bat:17` 直接执行 `python -c` 探测。Win10/11 未装 Python 时，`python` 会命中 `WindowsApps` 应用别名，**弹出 Microsoft Store**
3. 没有检查 `requirements.txt` 是否存在，缺文件时 pip 抛英文 `Could not open requirements file`，与整体中文提示风格断裂。
4. `install.bat:45` 的 pip 调用没加 `--no-input`，遇到证书/凭据确认会**停在交互提示上，表现为"卡死"**
5. 没有先确认 pip 是否可用，`ensurepip` 只写在事后补救的报错里。
6. 未说明编码要求：脚本用了 `chcp 65001`，但如果文件存成 GBK，中文提示全乱码；**存成带 BOM 的 UTF-8 时首行 `@echo off` 会被 BOM 破坏**
7. 全局安装落到 `Program Files` 下的 Python 时会因权限失败，报错里没点明"以管理员运行 / `--user`"

**run.bat**
1. `run.bat:51` 的 `for` 循环用 `set` 覆盖赋值，**只保留最后一个 PID**，多进程监听同一端口时提示不准。
2. `netstat -ano` 没加 `-p TCP`，把 UDP 行也捞进来了，靠 `LISTENING` 过滤属补救。
3. `run.bat:42-45` 的 `copy ... >nul` 失败了被吞掉，仍然打印"已生成 .env"——这是和缺陷 2 同类的**静默假成功**
4. 没有校验 `main.py` 是否存在，要等 uvicorn 抛 `Error loading ASGI app` 才暴露，且此时窗口已进入错误流
5. 没提示 `taskkill /F` 对系统进程/他人会话**需要管理员权限**
6. `:8000 ` 的匹配只防了 `80000`，未限 TCP，也没处理多 PID
7. 解释器探测逻辑与 install.bat 靠复制粘贴保持一致，任一处改动就会错位。

## 二、改进版脚本

**install.bat**

```bat
@echo off
rem ============================================================
rem  依赖安装脚本 install.bat
rem  要求：Windows + Python 3.10 及以上
rem  重要：本文件必须保存为 UTF-8（不带 BOM），否则 chcp 65001 下中文会乱码
rem ============================================================
chcp 65001 >nul
cd /d "%~dp0"
setlocal

rem ---------- 0.1 依赖清单是否存在 ----------
if not exist "requirements.txt" (
    echo [错误] 当前目录找不到 requirements.txt
    echo        当前目录：%CD%
    pause
    exit /b 1
)

rem ---------- 0.2 定位解释器：先过版本门槛，再认定 ----------
rem PY     = 最终选定的、版本满足 3.10 的解释器
rem PY_ANY = 找到但不一定满足版本的解释器，只用于给出准确报错
set "PY="
set "PY_ANY="

rem 优先 py 启动器：-3 只选 Python 3，再单独校验小版本是否 >= 3.10
py -3 -c "import sys" >nul 2>&1
if not errorlevel 1 (
    set "PY_ANY=py -3"
    py -3 -c "import sys; sys.exit(0 if sys.version_info[:2] >= (3, 10) else 1)" >nul 2>&1
    if not errorlevel 1 set "PY=py -3"
)

rem py 不可用或版本不够时，回退 PATH 中的 python
rem 先用 where 确认存在，避免命中 WindowsApps 应用别名而弹出 Microsoft Store
if not defined PY (
    where python >nul 2>&1
    if not errorlevel 1 (
        set "PY_ANY=python"
        python -c "import sys; sys.exit(0 if sys.version_info[:2] >= (3, 10) else 1)" >nul 2>&1
        if not errorlevel 1 set "PY=python"
    )
)

rem 三种结局分开处理，避免"版本过低"被误报成"未安装"
if defined PY goto :python_ready
if defined PY_ANY goto :version_too_low
goto :python_missing

:version_too_low
rem for 放在分支里而不是 if 括号内，防止复合语句把括号解析打断
for /f "delims=" %%v in ('%PY_ANY% -c "import sys;print(sys.version.split()[0])"') do set "PYVER=%%v"
echo [错误] 当前 Python 版本为 %PYVER%，低于本项目要求的 3.10。
echo        请升级 Python 后重新运行本脚本。
pause
exit /b 1

:python_missing
echo [错误] 未找到任何可用的 Python 解释器。
echo        请安装 Python 3.10+，安装时务必勾选 "Add Python to PATH"。
echo        下载地址：https://www.python.org/downloads/windows/
pause
exit /b 1

:python_ready
echo [info] 使用解释器：%PY%
rem 打印真实可执行文件路径，便于和 run.bat 对照，排查"装到别的解释器"
%PY% -c "import sys; print('       ' + sys.executable)"

rem ---------- 0.3 pip 是否可用 ----------
%PY% -m pip --version >nul 2>&1
if errorlevel 1 (
    echo [错误] 该解释器没有可用的 pip，请先修复：
    echo        %PY% -m ensurepip --upgrade
    pause
    exit /b 1
)

rem ---------- 1. 安装依赖 ----------
echo.
echo [1/2] 正在安装依赖：fastapi / uvicorn / openai / python-dotenv ...
rem 不要用 >nul 吞掉输出，pip 的报错信息是排查问题的唯一线索
rem --no-input 防止 pip 停在交互式输入上导致窗口看似卡死
%PY% -m pip install --disable-pip-version-check --no-input -r requirements.txt
if errorlevel 1 goto :pip_failed

rem ---------- 2. 导入自检 ----------
rem 只信 pip 返回码不够：可能装进别的解释器，也可能只装了一半
echo [2/2] 正在校验依赖能否导入 ...
%PY% -c "import fastapi, uvicorn, openai, dotenv" >nul 2>&1
if errorlevel 1 goto :import_failed

echo.
echo 安装完成！
echo   启动：双击 run.bat，浏览器打开 http://127.0.0.1:8000
echo   如需接入真实模型：把 .env.example 复制为 .env 并填写 API Key
pause
exit /b 0

:pip_failed
echo.
echo [错误] 依赖安装失败（pip 返回非 0 退出码）。
echo        请按顺序排查：
echo        1. 网络问题，换国内镜像重试：
echo           %PY% -m pip install -r requirements.txt -i https://pypi.tuna.tsinghua.edu.cn/simple
echo        2. pip 自身损坏，修复后重试：
echo           %PY% -m ensurepip --upgrade
echo        3. 权限不足（Python 装在 Program Files 下时需要管理员权限）：
echo           右键以管理员身份运行本脚本，或改用 --user 参数安装
pause
exit /b 1

:import_failed
echo.
echo [错误] 依赖安装看似成功，但导入自检失败。
echo        常见原因：install.bat 与 run.bat 用的不是同一套 Python。
echo        请执行 %PY% -c "import sys;print(sys.executable)" 核对路径。
pause
exit /b 1
```

**run.bat**

```bat
@echo off
rem ============================================================
rem  服务启动脚本 run.bat
rem  重要：本文件必须保存为 UTF-8（不带 BOM）
rem ============================================================
chcp 65001 >nul
cd /d "%~dp0"
setlocal enabledelayedexpansion

rem 端口集中在这里配置，被占用时改这一行即可
set "PORT=8000"
set "APP=main:app"

echo ============================================================
echo  步骤 1：环境自检
echo ============================================================

rem ---------- 1.1 源码是否就在脚本目录 ----------
if not exist "main.py" (
    echo [错误] 当前目录找不到 main.py
    echo        当前目录：%CD%
    echo        请把 run.bat 与源码放在同一目录后再运行。
    pause
    exit /b 1
)

rem ---------- 1.2 定位解释器：逻辑必须与 install.bat 完全一致 ----------
set "PY="
py -3 -c "import sys" >nul 2>&1
if not errorlevel 1 (
    py -3 -c "import sys; sys.exit(0 if sys.version_info[:2] >= (3, 10) else 1)" >nul 2>&1
    if not errorlevel 1 set "PY=py -3"
)
if not defined PY (
    where python >nul 2>&1
    if not errorlevel 1 (
        python -c "import sys; sys.exit(0 if sys.version_info[:2] >= (3, 10) else 1)" >nul 2>&1
        if not errorlevel 1 set "PY=python"
    )
)
if not defined PY (
    echo [错误] 未找到 3.10 及以上的 Python。
    echo        请先双击 install.bat 完成环境准备。
    pause
    exit /b 1
)

rem ---------- 1.3 依赖是否完整 ----------
rem 提前拦截，避免 uvicorn 抛 ModuleNotFoundError 让用户看不懂
%PY% -c "import uvicorn, fastapi" >nul 2>&1
if errorlevel 1 (
    echo [错误] 依赖未安装完整，请先双击运行 install.bat。
    pause
    exit /b 1
)

rem ---------- 1.4 首次运行生成 .env ----------
rem copy 失败必须被发现，不能像旧版那样吞掉返回码还打印成功
if not exist ".env" (
    if exist ".env.example" (
        copy ".env.example" ".env" >nul 2>&1
        if errorlevel 1 (
            echo [警告] 生成 .env 失败，将使用默认配置（样例回放模式）。
        ) else (
            echo [info] 已从 .env.example 生成 .env（样例回放模式，无需 API Key）。
        )
    ) else (
        echo [警告] 未找到 .env.example，跳过 .env 生成。
    )
)

rem ---------- 2. 启动前检查端口占用 ----------
rem -p TCP 只取 TCP 连接，排除 UDP 干扰
rem ":8000 " 带尾随空格精确匹配本地地址列，避免把 18000 / 80000 误判为占用
rem 只统计 LISTENING 行，PID 是第 5 列；多个进程监听同一端口时全部收集
set "PIDS="
for /f "tokens=5" %%p in ('netstat -ano -p TCP ^| findstr /R /C:":%PORT% .*LISTENING"') do (
    set "PIDS=!PIDS!%%p "
)
if defined PIDS goto :port_busy

rem ---------- 3. 启动服务 ----------
echo [info] 服务地址：http://127.0.0.1:%PORT%
echo [info] 按 Ctrl+C 停止服务
%PY% -m uvicorn %APP% --host 127.0.0.1 --port %PORT%
rem 非 0 退出码保留窗口方便看报错；手动 Ctrl+C 也可能进入这里，可忽略
if errorlevel 1 (
    echo.
    echo [错误] 服务异常退出（若是手动 Ctrl+C 可以忽略本提示）。
    echo        请查看上方 uvicorn 的输出。
    pause
    exit /b 1
)
exit /b 0

:port_busy
echo.
echo [错误] 端口 %PORT% 已被占用，占用进程 PID：!PIDS!
echo.
rem 占用的往往是上一轮没关掉的 uvicorn / python 进程
echo        查看是哪个程序：       tasklist /FI "PID eq <上面的PID>"
echo        结束该进程（需管理员）：taskkill /PID <上面的PID> /F
echo        或把本脚本顶部的 PORT 改成其它端口（如 8001）。
pause
exit /b 1
```
